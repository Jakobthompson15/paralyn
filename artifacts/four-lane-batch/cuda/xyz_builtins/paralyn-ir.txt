kernel record_coordinates(output: read_write buffer<i32>, guard: i32) {
  let x:i32 = i32(((blockIdx.x * blockDim.x) + threadIdx.x))
  let y:i32 = i32(((blockIdx.y * blockDim.y) + threadIdx.y))
  let z:i32 = i32(((blockIdx.z * blockDim.z) + threadIdx.z))
  let width:i32 = i32((gridDim.x * blockDim.x))
  let height:i32 = i32((gridDim.y * blockDim.y))
  let base:i32 = ((((((z * height) + y) * width) + x) * 12:i32) + guard)
  store output[(base + 0:i32)] = i32(threadIdx.x)
  store output[(base + 1:i32)] = i32(threadIdx.y)
  store output[(base + 2:i32)] = i32(threadIdx.z)
  store output[(base + 3:i32)] = i32(blockIdx.x)
  store output[(base + 4:i32)] = i32(blockIdx.y)
  store output[(base + 5:i32)] = i32(blockIdx.z)
  store output[(base + 6:i32)] = i32(blockDim.x)
  store output[(base + 7:i32)] = i32(blockDim.y)
  store output[(base + 8:i32)] = i32(blockDim.z)
  store output[(base + 9:i32)] = i32(gridDim.x)
  store output[(base + 10:i32)] = i32(gridDim.y)
  store output[(base + 11:i32)] = i32(gridDim.z)
}
