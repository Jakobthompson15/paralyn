cmake_minimum_required(VERSION 3.24)
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "26.0" CACHE STRING "Minimum deployment target (qualified LLVM bottle requires macOS 26)")
endif()
project(Paralyn VERSION 0.0.1 LANGUAGES C CXX)
execute_process(COMMAND git -C "${PROJECT_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE PARALYN_BUILD_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT PARALYN_BUILD_COMMIT)
  set(PARALYN_BUILD_COMMIT "unknown")
endif()
execute_process(COMMAND git -C "${PROJECT_SOURCE_DIR}" status --porcelain --untracked-files=normal OUTPUT_VARIABLE PARALYN_BUILD_CHANGES ERROR_QUIET)
if(PARALYN_BUILD_CHANGES)
  set(PARALYN_BUILD_DIRTY "true")
else()
  set(PARALYN_BUILD_DIRTY "false")
endif()
include(GNUInstallDirs)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(PARALYN_ENABLE_METAL "Build physical Metal execution backend" ${APPLE})
option(PARALYN_BUILD_COMPILER "Build Clang CUDA frontend (runtime-only installations do not need LLVM)" ${APPLE})
if(PARALYN_ENABLE_METAL AND NOT APPLE)
  message(FATAL_ERROR "Metal requires macOS. CUDA and HIP backends are not implemented yet.")
endif()
if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(paralyn_ir STATIC compiler/ir/ir.cpp compiler/ir/artifact.cpp compiler/codegen/metal.cpp)
target_include_directories(paralyn_ir PUBLIC $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_sources(paralyn_ir PRIVATE compiler/ir/executable.cpp)
if(PARALYN_ENABLE_METAL)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  add_library(paralyn_runtime STATIC backends/metal/runtime.mm backends/metal/engine.mm)
  target_compile_options(paralyn_runtime PRIVATE -fobjc-arc)
  target_link_libraries(paralyn_runtime PUBLIC paralyn_ir "-framework Metal" "-framework Foundation")
else()
  add_library(paralyn_runtime STATIC backends/unavailable.cpp)
  target_link_libraries(paralyn_runtime PUBLIC paralyn_ir)
endif()
target_sources(paralyn_runtime PRIVATE runtime/telemetry.cpp)
target_include_directories(paralyn_runtime PUBLIC include)
add_library(paralyn_native SHARED runtime/native.cpp)
target_include_directories(paralyn_native PUBLIC $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(paralyn_native PRIVATE paralyn_runtime)
set_target_properties(paralyn_native PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
set(LLVM_PACKAGE_VERSION "not installed (runtime-only build)")
if(PARALYN_BUILD_COMPILER)
  find_package(LLVM CONFIG REQUIRED HINTS /opt/homebrew/opt/llvm@21/lib/cmake/llvm)
  find_package(Clang CONFIG REQUIRED HINTS "${LLVM_INSTALL_PREFIX}/lib/cmake/clang")
  set(PARALYN_CLANG_PATH "${LLVM_TOOLS_BINARY_DIR}/clang++")
  execute_process(COMMAND "${PARALYN_CLANG_PATH}" -print-resource-dir OUTPUT_VARIABLE PARALYN_RESOURCE_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  set(PARALYN_SDK_PATH "")
  if(APPLE)
    execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE PARALYN_SDK_PATH OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  endif()
  message(STATUS "Paralyn frontend: LLVM ${LLVM_PACKAGE_VERSION} at ${LLVM_INSTALL_PREFIX}")
  add_library(paralyn_frontend STATIC compiler/frontend/frontend.cpp)
  target_include_directories(paralyn_frontend SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
  target_include_directories(paralyn_frontend PUBLIC include)
  target_link_libraries(paralyn_frontend PUBLIC paralyn_ir PRIVATE clang-cpp LLVM)
  target_compile_definitions(paralyn_frontend PRIVATE PARALYN_INCLUDE_DIR="${PROJECT_SOURCE_DIR}/include"
    PARALYN_CLANG_PATH="${PARALYN_CLANG_PATH}" PARALYN_RESOURCE_DIR="${PARALYN_RESOURCE_DIR}" PARALYN_SDK_PATH="${PARALYN_SDK_PATH}")
  add_library(unicuda_frontend ALIAS paralyn_frontend)
endif()
if(WIN32)
  set(PARALYN_PROCESS_SOURCE cli/process_windows.cpp)
else()
  set(PARALYN_PROCESS_SOURCE cli/process_posix.cpp)
endif()
add_executable(paralyn cli/main.cpp ${PARALYN_PROCESS_SOURCE})
target_include_directories(paralyn SYSTEM PRIVATE third_party)
target_link_libraries(paralyn PRIVATE paralyn_native)
if(NOT PARALYN_BUILD_COMPILER)
  target_link_libraries(paralyn PRIVATE paralyn_ir)
endif()
if(PARALYN_BUILD_COMPILER)
  target_link_libraries(paralyn PRIVATE paralyn_frontend)
endif()
target_compile_definitions(paralyn PRIVATE
  PARALYN_HAS_COMPILER=$<BOOL:${PARALYN_BUILD_COMPILER}> PARALYN_HAS_METAL=$<BOOL:${PARALYN_ENABLE_METAL}>
  PARALYN_SOURCE_DIR="${PROJECT_SOURCE_DIR}" PARALYN_BINARY_DIR="${PROJECT_BINARY_DIR}"
  PARALYN_INCLUDE_DIR="${PROJECT_SOURCE_DIR}/include" PARALYN_HOST_CXX="${CMAKE_CXX_COMPILER}" PARALYN_HOST_CC="${CMAKE_C_COMPILER}"
  PARALYN_DEPLOYMENT_TARGET="${CMAKE_OSX_DEPLOYMENT_TARGET}" PARALYN_RUNTIME_ARCHIVE="$<TARGET_FILE:paralyn_runtime>"
  PARALYN_NATIVE_LIBRARY="$<TARGET_FILE:paralyn_native>" PARALYN_IR_ARCHIVE="$<TARGET_FILE:paralyn_ir>"
  PARALYN_LLVM_VERSION="${LLVM_PACKAGE_VERSION}" PARALYN_BUILD_COMMIT="${PARALYN_BUILD_COMMIT}" PARALYN_BUILD_DIRTY="${PARALYN_BUILD_DIRTY}")
if(UNIX)
  add_custom_command(TARGET paralyn POST_BUILD COMMAND ${CMAKE_COMMAND} -E create_symlink paralyn ${PROJECT_BINARY_DIR}/unicuda)
else()
  add_custom_command(TARGET paralyn POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:paralyn> ${PROJECT_BINARY_DIR}/unicuda.exe)
endif()
add_library(unicuda_ir ALIAS paralyn_ir)
add_library(unicuda_runtime ALIAS paralyn_runtime)
if(APPLE)
  set_target_properties(paralyn PROPERTIES INSTALL_RPATH "@loader_path/../lib")
elseif(UNIX)
  set_target_properties(paralyn PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
endif()
install(TARGETS paralyn paralyn_native paralyn_ir paralyn_runtime RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY docs/ DESTINATION share/paralyn/docs)
install(FILES README.md LICENSE THIRD_PARTY_LICENSES.md DESTINATION share/paralyn)
install(DIRECTORY bindings/python/paralyn DESTINATION share/paralyn/python PATTERN __pycache__ EXCLUDE)
install(FILES $<TARGET_FILE:paralyn_native> DESTINATION share/paralyn/python/paralyn/_native)
install(DIRECTORY third_party/ DESTINATION share/paralyn/third_party)
if(UNIX)
  install(CODE "execute_process(COMMAND \"${CMAKE_COMMAND}\" -E create_symlink paralyn \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/bin/unicuda\")")
endif()
if(PARALYN_BUILD_COMPILER)
  add_custom_command(OUTPUT ${PROJECT_BINARY_DIR}/native-kernels.prk
    COMMAND ${CMAKE_COMMAND} -E rm -f ${PROJECT_BINARY_DIR}/native-kernels.prk
    COMMAND $<TARGET_FILE:paralyn> compile ${PROJECT_SOURCE_DIR}/examples/native/kernels.cu --output ${PROJECT_BINARY_DIR}/native-kernels.prk
    DEPENDS paralyn ${PROJECT_SOURCE_DIR}/examples/native/kernels.cu)
  add_custom_target(native_kernels ALL DEPENDS ${PROJECT_BINARY_DIR}/native-kernels.prk)
  if(EXISTS "${PROJECT_SOURCE_DIR}/bindings/python/operators.cu")
    add_custom_command(OUTPUT ${PROJECT_BINARY_DIR}/operators.prk
      COMMAND ${CMAKE_COMMAND} -E rm -f ${PROJECT_BINARY_DIR}/operators.prk
      COMMAND $<TARGET_FILE:paralyn> compile ${PROJECT_SOURCE_DIR}/bindings/python/operators.cu --output ${PROJECT_BINARY_DIR}/operators.prk
      DEPENDS paralyn ${PROJECT_SOURCE_DIR}/bindings/python/operators.cu)
    add_custom_target(array_operators ALL DEPENDS ${PROJECT_BINARY_DIR}/operators.prk)
    install(FILES ${PROJECT_BINARY_DIR}/operators.prk DESTINATION share/paralyn/python/paralyn/_data)
    install(FILES ${PROJECT_BINARY_DIR}/operators.prk DESTINATION share/paralyn)
  endif()
endif()
if(BUILD_TESTING)
  add_executable(ir_tests tests/ir_tests.cpp)
  target_link_libraries(ir_tests PRIVATE paralyn_ir)
  add_test(NAME ir_and_codegen COMMAND ir_tests)
  add_executable(artifact_tests tests/artifact_tests.cpp)
  target_link_libraries(artifact_tests PRIVATE paralyn_ir)
  add_test(NAME kernel_artifacts COMMAND artifact_tests)
  if(EXISTS "${PROJECT_SOURCE_DIR}/tests/executable_tests.cpp")
    add_executable(executable_tests tests/executable_tests.cpp)
    target_link_libraries(executable_tests PRIVATE paralyn_ir)
    add_test(NAME executable_artifacts COMMAND executable_tests)
  endif()
  if(PARALYN_BUILD_COMPILER)
    add_executable(frontend_tests tests/frontend_tests.cpp)
    target_link_libraries(frontend_tests PRIVATE paralyn_frontend)
    target_compile_definitions(frontend_tests PRIVATE PARALYN_TEST_SOURCE_DIR="${PROJECT_SOURCE_DIR}")
    add_test(NAME frontend COMMAND frontend_tests)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME native_compile COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/compile_cli.py --paralyn $<TARGET_FILE:paralyn> --source ${PROJECT_SOURCE_DIR}/examples/native/kernels.cu)
  endif()
  if(PARALYN_ENABLE_METAL)
    add_executable(runtime_smoke tests/runtime_smoke.cpp)
    target_link_libraries(runtime_smoke PRIVATE paralyn_runtime)
    add_test(NAME metal_runtime COMMAND runtime_smoke)
    set_tests_properties(metal_runtime PROPERTIES LABELS "gpu")
    add_test(NAME unobserved_runtime_failure COMMAND ${CMAKE_COMMAND} -DRUNTIME=$<TARGET_FILE:runtime_smoke> -P ${PROJECT_SOURCE_DIR}/tests/runtime_failure.cmake)
    set_tests_properties(unobserved_runtime_failure PROPERTIES LABELS "gpu;gate-b" TIMEOUT 30)
    if(PARALYN_BUILD_COMPILER)
      add_executable(native_c examples/native/vector_add.c)
      add_executable(native_cpp examples/native/vector_add.cpp)
      add_executable(native_tests tests/native/native_tests.cpp)
      foreach(target native_c native_cpp native_tests)
        target_link_libraries(${target} PRIVATE paralyn_native)
        add_dependencies(${target} native_kernels)
        target_compile_options(${target} PRIVATE -fno-fast-math -ffp-contract=off)
      endforeach()
      set_property(TARGET native_c PROPERTY C_STANDARD 11)
      foreach(kind native-c native-cpp native-tests python-tests python-example python-finalizer)
        string(REPLACE "-" "_" test_name ${kind})
        add_test(NAME ${test_name} COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/native/run_example.py --kind ${kind} --build ${PROJECT_BINARY_DIR})
        set_tests_properties(${test_name} PROPERTIES LABELS "gpu;native" TIMEOUT 120)
      endforeach()
      add_test(NAME gate_a COMMAND ${CMAKE_COMMAND} -DPYTHON=${Python3_EXECUTABLE} -DSOURCE_DIR=${PROJECT_SOURCE_DIR} -DPARALYN=$<TARGET_FILE:paralyn> -P ${PROJECT_SOURCE_DIR}/tests/gate_a.cmake)
      set_tests_properties(gate_a PROPERTIES LABELS "gpu;gate-a" TIMEOUT 120)
      add_test(NAME gate_b_correctness COMMAND ${CMAKE_COMMAND} -DPYTHON=${Python3_EXECUTABLE} -DSOURCE_DIR=${PROJECT_SOURCE_DIR} -DPARALYN=$<TARGET_FILE:paralyn> -P ${PROJECT_SOURCE_DIR}/tests/gate_b.cmake)
      set_tests_properties(gate_b_correctness PROPERTIES LABELS "gpu;gate-b" TIMEOUT 300)
      add_executable(gate_b_benchmark benchmarks/gate_b_benchmark.cpp)
      target_link_libraries(gate_b_benchmark PRIVATE paralyn_frontend paralyn_runtime)
      target_compile_options(gate_b_benchmark PRIVATE -O2 -fno-fast-math -ffp-contract=off)
    endif()
  endif()
endif()
